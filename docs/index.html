Pebble Playground

Pebble is a tiny programming language, built from scratch: a tokenizer, a parser and an interpreter. Write code below and press Run.


    

    

  


  

    

      Language cheat sheet

let x = 5                 # variable
x = x + 1                 # change it
let xs = [1, 2, 3]        # list; xs[0] = 9
print("hi", x)            # output

if x > 3 { print("big") } else { print("small") }
while x < 10 { x = x + 1 }
for item in [1, 2, 3] { print(item) }   # also range(5)

fn square(n) { return n * n }
let add = fn(a, b) { return a + b }     # functions are values

and  or  not   == != < <= > >=   + - * / %
builtins: print len str num type range push pop sort reverse
          join split upper lower abs floor sqrt min max
Statements end with a new line or ;   Comments start with #
